```python
import tkinter as tk
from tkinter import filedialog
import pygame

# 初始化pygame
pygame.mixer.init()

# 创建播放器窗口
root = tk.Tk()
root.title("Python 音频播放器")
root.geometry("400x200")

# 播放器的状态
is_playing = False
current_song = None

# 播放音频的函数
def play_song():
    global is_playing, current_song
    if current_song:
        pygame.mixer.music.load(current_song)
        pygame.mixer.music.play()
        is_playing = True
        play_button.config(text="暂停")

# 暂停音频的函数
def pause_song():
    global is_playing
    if is_playing:
        pygame.mixer.music.pause()
        play_button.config(text="播放")
        is_playing = False
    else:
        pygame.mixer.music.unpause()
        play_button.config(text="暂停")
        is_playing = True

# 停止音频的函数
def stop_song():
    global is_playing
    pygame.mixer.music.stop()
    play_button.config(text="播放")
    is_playing = False

# 打开文件对话框并加载音频
def open_file():
    global current_song
    current_song = filedialog.askopenfilename(filetypes=[("音频文件", "*.mp3 *.wav *.ogg")])
    if current_song:
        play_song()

# 创建按钮
play_button = tk.Button(root, text="播放", command=pause_song)
play_button.pack(pady=20)

stop_button = tk.Button(root, text="停止", command=stop_song)
stop_button.pack(pady=5)

open_button = tk.Button(root, text="打开", command=open_file)
open_button.pack(pady=5)

# 启动主循环
root.mainloop()
```
